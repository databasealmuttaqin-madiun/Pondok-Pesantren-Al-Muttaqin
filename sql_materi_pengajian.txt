-- Tabel materi_pengajian
CREATE TABLE public.materi_pengajian (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    kelompok TEXT NOT NULL CHECK (kelompok IN ('alquran', 'himpunan')),
    nama_materi TEXT NOT NULL,
    jumlah_halaman INTEGER NOT NULL DEFAULT 0,
    urutan INTEGER NOT NULL DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Mengaktifkan Row Level Security
ALTER TABLE public.materi_pengajian ENABLE ROW LEVEL SECURITY;

-- Policy untuk SELECT (Semua user yang login bisa membaca)
CREATE POLICY "Enable read access for authenticated users" 
ON public.materi_pengajian FOR SELECT 
TO authenticated USING (true);

-- Policy untuk INSERT, UPDATE, DELETE (Hanya bisa dilakukan dari API/Role tertentu)
-- Catatan: RLS detail ini bisa disesuaikan dengan skema tabel users/role yang Anda miliki
CREATE POLICY "Enable ALL access for admins" 
ON public.materi_pengajian FOR ALL 
TO authenticated 
USING (true); -- Hati-hati, jika Anda menggunakan tabel roles, ganti ini dengan auth.jwt()->>'role' = 'admin' dsb.
