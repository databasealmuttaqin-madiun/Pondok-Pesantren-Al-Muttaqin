-- Tabel target_pengajian
CREATE TABLE IF NOT EXISTS public.target_pengajian (
    id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    kelas_pengajian TEXT NOT NULL,
    materi_id BIGINT NOT NULL REFERENCES public.materi_pengajian(id) ON DELETE CASCADE,
    tanggal DATE NOT NULL,
    pertemuan_ke INTEGER NOT NULL DEFAULT 1,
    target_halaman_mulai INTEGER NOT NULL,
    target_halaman_selesai INTEGER NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

-- Mengaktifkan Row Level Security
ALTER TABLE public.target_pengajian ENABLE ROW LEVEL SECURITY;

-- Policy untuk SELECT (Semua user yang login bisa membaca)
DROP POLICY IF EXISTS "Enable read access for authenticated users" ON public.target_pengajian;
CREATE POLICY "Enable read access for authenticated users" 
ON public.target_pengajian FOR SELECT 
TO authenticated USING (true);

-- Policy untuk INSERT, UPDATE, DELETE 
-- Memberikan akses (Karena role checking dilakukan di UI/Frontend)
DROP POLICY IF EXISTS "Enable all access for everyone" ON public.target_pengajian;
CREATE POLICY "Enable all access for everyone" 
ON public.target_pengajian FOR ALL 
USING (true)
WITH CHECK (true);

-- SEED DATA Target Capaian (Contoh)
-- Menggunakan subquery untuk mengambil materi_id yang valid
INSERT INTO public.target_pengajian (kelas_pengajian, materi_id, tanggal, pertemuan_ke, target_halaman_mulai, target_halaman_selesai)

FROM public.materi_pengajian WHERE nama_materi = 'Juz 1' LIMIT 1;
